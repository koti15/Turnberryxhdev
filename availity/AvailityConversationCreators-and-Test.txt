public with sharing class AvailityHostConversationCreator extends AvailityMessageProcessor {



    private static final String QUEUE_NAME = 'Availity Provider ITS Host';

    private static final String RECORD_TYPE = 'Open Case';



    /\* =========================

       Owner Resolution

       ========================= \*/

    public override Id getOwner() {



        List\<Group> queue = [

            SELECT Id FROM Group WHERE Type = 'Queue' AND Name = :QUEUE_NAME LIMIT 1 ];

        if (queue.isEmpty()) {

            throw new IllegalArgumentException('Queue not found: ' + QUEUE_NAME);

        }

        return queue[0].Id;

    }





    /\* =========================

       Host Contact Resolution

       ========================= \*/

    public override Contact getOrCreateContact() {



        String memberId = (String) payload.get('patientId');

        String numericId = memberId.replaceAll('[^0-9]', '');



        List\<Contact> contacts = [SELECT Id , AccountId FROM Contact WHERE Member_Id__c = :numericId OR AGP_ID__c = :numericId LIMIT 2];



        if (contacts.size() == 1) {

            return contacts[0];

        }



        if (contacts.size() > 1) {

            throw new IllegalArgumentException('Multiple Host contacts found for patientId: ' + memberId);

        }



        throw new IllegalArgumentException('Host member not found for patientId: ' + memberId);

    }

}

public with sharing class AvailityCareConversationCreator extends AvailityMessageProcessor{



 private static final string QUEUE_NAME = 'Availity Provider Message';

    Private static final String RECORD_TYPE = 'Open Case';



    Public override id getOwner(){      

        List\<Group> queue = [select id from Group where Type = 'Queue' AND Name =: QUEUE_NAME Limit 1];

        if(queue.isEmpty()){

            throw new IllegalArgumentException('Queue Not Found:');            

        }

        return queue[0].id;

    }





    Public override Contact getOrCreateContact(){  &#x20;



        String memberId = (String) payload.get('patientId');

        String numericId = memberId.replaceAll('[^0-9]', '');



        List \<Account> accounts = [select id, PersonContactId from account where

                                   IsPersonAccount = true and Member_Id__c = :numericId Limit 2];

        if(accounts.size() == 1){

            return  new Contact(Id = accounts[0].PersonContactId);

        }



        throw new IllegalArgumentException('Host member not found - Personaccount creation pending:' +numericId);

    }



}
