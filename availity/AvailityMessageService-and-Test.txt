        AvailityMessageProcessor processor;

    private static void handleMessageLegacy() 
    {
        // Get request body
        try{
            String caseDescription = '';
            RestRequest req = RestContext.request;
            String body = req.requestBody.toString();
            
            if (body == null || String.isBlank(body)) {
                JsonResponse.failure('Empty request body');
                return;
            }
           // CNC_HandleCustomException.customExceptionForBedlam('pusher_message_service', 'pusherreqbody', body);
            // Parse JSON
            Map<String, Object> payload;
            try {
                payload = (Map<String, Object>) JSON.deserializeUntyped(body);
            } catch (Exception e) {
                JsonResponse.failure('Invalid JSON format');
                return;
            }
            
            String createdDate = (String) payload.get('createDate');
            String conversationId = (String) payload.get('id');
            List<Object> messages = (List<Object>) payload.get('messages');
            
            if(String.isEmpty(createdDate) || String.isEmpty(conversationId) || messages == null || messages.size() == 0){
                JsonResponse.failure('Hey something went wrong with json request. please check fields such as id, createDate and messages. Thankyou.');
                return;
            }
            
            
            // Start main logic
            Conversation__c conversation;
            List<Conversation__c> existingConversations = [
                SELECT Id, Claim_Number__c, Conversation_Status__c, Contact__c FROM Conversation__c WHERE Availity_Unique_ID__c = :conversationId LIMIT 1
            ];
            
            String existingConversationStatus = '';
            if (existingConversations.isEmpty()) {
                // No existing Conversation - create new
                try{
                    
                    String memberId = ((String) payload.get('patientId'));
                    if(String.isEmpty(memberId)){
                        throw new IllegalArgumentException('Patient Id is Empty');
                    }
                    conversation = createConversation(payload, memberId.replaceAll('[^0-9]', ''));
                    
                    if(conversation.Claim_Number__c != null){
                        caseDescription = conversation.Claim_Number__c;
                        conversation.Claim_Number__c = conversation.Claim_Number__c.split(',')[0];
                    }
                    System.debug('------caseDescription--' + caseDescription);
                    insert conversation;
                }catch(IllegalArgumentException e){
                    System.debug('exception message ' + e);
                    JsonResponse.failure(
                        e.getMessage()
                    );
                    return;
                }
            } else {
                existingConversationStatus = existingConversations[0].Conversation_Status__c;
                conversation = existingConversations[0];
                updateInfoOnAconversation(payload,conversation);
            }
            
            // Check for latest Case linked to this Conversation
            Case targetCase;
            List<Case> relatedCases = [
                SELECT Id, Status, ContactId FROM Case
                WHERE Conversation__c = :conversation.Id
                ORDER BY CreatedDate DESC
                LIMIT 1
            ];
            
            if (relatedCases.isEmpty()) {
            // No Case → create new and passing memberId
               
                targetCase = createNewCase(conversation, createdDate,caseDescription );
            } else {
                Case existingCase = relatedCases[0];
                //if (existingCase.Status == 'Closed') {
                
                if (conversation.Conversation_Status__c == 'REOPENED' && existingConversationStatus != 'REOPENED' ) {
                    // Closed → create new
                    targetCase = cloneClosedCaseAsNew(existingCase.Id,conversation.Claim_Number__c);
                } else {
                    // Use existing open Case
                    targetCase = existingCase;
                }
            }
            
            // Create new Message
            List<Id> createdMessageIds = new List<Id>();
            for(Object messageObject : messages){
                try{
                    Map<String,Object> message = (Map<String,Object>) messageObject;
                    Map<String,Object> user = (Map<String,Object>) message.get('user');
                    combineTwoMaps(message,user);
                    Message__c newMessage = createMessage(message,conversationId);
                    newMessage.Conversation__c = conversation.Id;
                    insert newMessage;
                    createdMessageIds.add(newMessage.Id);
                } catch (Exception e){
                    JsonResponse.failure('Something went wrong while creating message object, ' + e.getMessage());
                    return;
                }
                
            }
            
            
            // Return success
            JsonResponse.success(
                New Map<String,Object>{
                    'message' => 'Message processed successfully',
                        'messageIds' => createdMessageIds,
                        'caseId' => targetCase.Id,
                        'conversationId' => conversation.Id
                        }
            );
        } catch (Exception e){
            JsonResponse.failure(e.getMessage());
        }
    }
