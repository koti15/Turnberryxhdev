        //TODO:  if no existing conversations - its a NEW_CONVERSATION, if the conversation exists and no case exists its a NEW_CONVERSATION_AND_CASE if the related conversation status is not == reopened and the incoming message status is reopened it's a REOPENED_CONVERSATION

    Public virtual void process(){
        try{
            validateMessage();
            
            this.relatedConversations = getRelatedConversations();
            this.relatedCases = getExistingCases();
            
            processingType = resolveProcessingType();
            
            switch on processingType {
                when NEW_CASE {
                    // Implementation for new conversation and case
                    targetConversation = relatedConversations[0];
                    createMessages();
                    targetCase         = createRelatedCase();
                    // also update conversation status if it has changed on this message 
                    applyConversationUpdates(targetConversation);
                }
                when NEW_CONVERSATION {
                    // Implementation for update conversation and case
                    targetConversation = createConversation();
                    createMessages();
                    targetCase         = createRelatedCase();
                    
                }
                 when REOPENED_CONVERSATION{
                    //implementation for the reopoend
                   // targetConversation = createConversation(); // new record, not update//commented by suram for 1816 
                     targetConversation = relatedConversations[0];
                     applyConversationUpdates(targetConversation);
                    createMessages();
                    targetCase = cloneCaseAsNew(relatedCases[0].Conversation__c,(String) payload.get('reason'));
                  }
                when OPEN_CONVERSATION {
                    // implementation for open conversation
                    targetConversation = relatedConversations[0];
                    createMessages();
                    targetCase         = relatedCases[0];
                    // We must apply any field updates from the incoming message. 
                    applyConversationUpdates(targetConversation);
                }
            }
            commitWork();
            if(!isError){
                buildSuccessResponse();
            } 
        }catch (IllegalArgumentException iae) {
            system.debug('line non 433');
            buildErrorResponse(400, iae.getMessage());
        } catch (Exception e) {
            System.debug(e);
            system.debug('stackTrace: \n' + e.getStackTraceString() );
            system.debug('Exception Message:  \n' +e.getMessage());
            //CS-1708
            //preserve legacy avality behaviour so the same request returns
            //the same error result regardless of the new processor toggle
            buildErrorResponse(400,e.getMessage());
        }
        
        
    }

* @description Resolves the Contact tied to this inbound message and returns it, throwing
*              IllegalArgumentException when resolution errors out or no match is found.
*              MEMOIZED: the underlying resolveMember() call (member resolution logic/service)
*              and the Contact SOQL query only run on the first invocation per transaction —
*              subsequent calls return the cached `cachedContact`. This method is called from
*              both createConversation() and createRelatedCase() in the NEW_CONVERSATION path,
*              so caching avoids doing the resolution + query twice for the same inbound message.
*/
    protected virtual Contact getOrCreateContact() {
        
        if (cachedContact != null) {
            return cachedContact;
        }
        
        MemberResolutionContext resolution =
            resolveMember();
        
        if (resolution.hasError()) {
            throw new IllegalArgumentException(
                resolution.error
            );
        }
        
        if (!resolution.isResolved()) {
            throw new IllegalArgumentException(
                'Contact reference for a matching patient was not found.'
            );
        }
        
        softSearchMemberId = resolution.softMemberId;
        
        cachedContact = [
            SELECT Id, AccountId
            FROM Contact
            WHERE Id = :resolution.resolvedContactId
            LIMIT 1
        ];
        
        return cachedContact;
    }

*              if it's new, inserts the Case for the NEW_CONVERSATION/NEW_CASE/REOPENED_CONVERSATION
*              paths (linking it to the conversation first), updates the Conversation__c,
*              links the new messages to the conversation, and inserts them.
*/
    protected virtual void commitWork(){       
        //todo insert or update the conversation and case - try/catch the dml process errors with build error response
        system.debug('check target conversation ### \n' + this.targetConversation);
        if (targetConversation.Id == null) {
            insert targetConversation;
        }
        
        if (processingType == AvailityMessageProcessor.ProcessingType.NEW_CONVERSATION ||
            processingType == AvailityMessageProcessor.ProcessingType.NEW_CASE ||
            processingType == AvailityMessageProcessor.ProcessingType.REOPENED_CONVERSATION) {
                
                
                system.debug('targetCase: ' + targetCase);
                system.debug('processingType ' + processingType);
                resolveCaseConversation();
                
                insert targetCase;
            }
        update targetConversation;
        resolveMessages();
        insert newMessages;
    }  
    
    /**
* @description Builds the list of new Message__c records (not yet inserted) from the
*              inbound `messages` payload array, one per entry, via createMessage().
*/
    protected virtual void createMessages(){
        if(this.newMessages == null){
            this.newMessages = new list<message__c>();
        }
        for(object messageObj:messages){
            newMessages.add(createMessage((map<string,object>)messageObj));
        }
        
        // process error handeling for empty fields;
        String errorMessage = processEmptyFileldNames();
        if(errorMessage != ''){
            System.debug('---error--' + errorMessage);
            throw new IllegalArgumentException(errorMessage);
        }
    }
    
    /**
* @description Helper for createMessage(): sets a field on the given sObject when the
*              source value is present (casting to Date when isDate is true), otherwise
*              records the field name in `emptyFields` so processEmptyFileldNames() can
*              surface which fields were missing on the inbound message.
*/
    protected void populateMessageOrTrackMissing(string fieldName, object value, sObject record,string fieldApiName, boolean isDate){
        if(value != null){
            if(isDate){
                record.put(fieldAPIName,date.valueOf((string)value));
            } else {
                record.put(fieldAPIName,value);
            }   
        } else {
            emptyFields.add(fieldName);
        }
    }
    
    /**
* @description Links every message built in createMessages() to the (now known)
*              targetConversation.Id. Must run after targetConversation has been
*              inserted/assigned, and before newMessages are inserted.
*/
    protected void resolveMessages(){
        for(message__c newMessage:newMessages){
            newMessage.Conversation__c = targetConversation.id;
        }
    }
    
    /**
* @description Links targetCase to targetConversation.Id. Must run after targetConversation
*              has been inserted/assigned, and before targetCase is inserted.
*/
    protected void resolveCaseConversation(){
        targetCase.Conversation__c = targetConversation?.id;
    }
    
    /**
* @description Maps a single inbound message object (one entry of the `messages` payload
*              array) onto a new Message__c record, using populateMessageOrTrackMissing()
*              field-by-field so any missing values are tracked rather than causing a null
*              pointer exception.
*/
    protected virtual Message__c createMessage(Map<String, Object> payload){
        
        // Flatten the nested "user" object into the message payload so its keys become
        // user_id / user_type / user_firstName / user_lastName — same helper Service uses
        // (AvailityMessageService.combineTwoMaps) before mapping fields below. Without this
        // step messageUserObject only has plain keys ('id','type', etc.), not 'user_id' etc.
        Map<String,Object> messageUserObject = (Map<String,Object>) payload.get('user');
        AvailityMessageService.combineTwoMaps(payload, messageUserObject);
        
        Message__c newMessage = new Message__c(Message_Source__c = 'Inbound');
        
        // define mapping for the message object — same field mapping used by
        // AvailityMessageService.createMessage(), plus claimControlNum -> Claim_Id__c
        // which is specific to this processor
        Map<String, String> fieldMapping = new Map<String,String>{
            'Availity_Unique_ID__c'       => 'id',
                'Availity_Conversation_ID__c' => 'conversationId',
                'Message_Text__c'             => 'message',
                'Create_ID__c'                => 'createId',
                'Create_Date__c'              => 'createDate',
                'Last_Update_Id__c'           => 'lastUpdateId',
                'Last_Update_Date__c'         => 'lastUpdateDate',
                'User_Id__c'                  => 'user_id',
                'User_Type__c'                => 'user_type',
                'User_First_Name__c'          => 'user_firstName',
                'User_Last_Name__c'           => 'user_lastName'
                };
                    
                    for(String fieldApi : fieldMapping.keySet()){
                        String payloadKey = fieldMapping.get(fieldApi);
                        Boolean isDate = fieldApi.contains('Date');
                        populateMessageOrTrackMissing(payloadKey, payload.get(payloadKey), newMessage, fieldApi, isDate);
                    }
        
        system.debug('#### \n' +'Sanity check: ' + newMessage);
        return newMessage;
    }
    
    /**
* @description Builds a human-readable error string listing any field names that were
*              tracked as missing/blank by populateMessageOrTrackMissing() while mapping
*              inbound messages.
*/
    protected String processEmptyFileldNames(){
        System.debug('---error--' + this.emptyFields);
        String error = '';
        if(emptyFields != null && emptyFields.size() > 1){
            error = 'Please check following fields, as they are either null or empty : ';
            String fields = String.join(this.emptyFields, ', ');
            error = error + ' ' + fields;
        } else if (this.emptyFields.size() == 1){
            error = emptyFields[0] + ' is null or empty. Please check :-)';
        }
        System.debug('---error--' + error);
        return error;
    }
