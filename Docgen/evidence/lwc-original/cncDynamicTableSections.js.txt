    setRecordDataFromEvent(data) {
        this._recordData = data.searchResult;
        if (this._recordData) {
            //console.log('entity name', this._recordData.sectionName);
            //This condition needs to be removed from code.It was addedto handle the duplicate data issue.
            if (this._recordData.sectionName === 'Attach Claims EntityNotValid') {
                if (this._recordData.responsedata && (Array.isArray(this._recordData.responsedata) && this._recordData.responsedata.length > 0 && this._recordData.responsedata.some(obj => Object.keys(obj).length != 0))) {
                    let caseId = this._recordData.caseId;
                    let data = [];
                    let filteredData = [];
                    data = this._recordData.responsedata;
                    getMemberId({ recordId: caseId }).then(result => {
                        let memberId = result;
                        data.forEach(element => {
                            if (element.memberId === memberId) {
                                filteredData = [element];
                            }
                        });
                        // console.log('filterdata ', JSON.stringify(this._recordData.responsedata));
                        if (filteredData.length > 0 && filteredData.some(obj => Object.keys(obj).length != 0)) {
                            // console.log('dss ', JSON.stringify(filteredData));
                            this._recordData.responsedata = filteredData;
                            //console.log('ddddd ', JSON.stringify(this._recordData.responsedata));
                            this.showTable = true;
                            this.initilizeRecordData();
                        } else {
                            this.isLoading = false;
                            this.showTable = false;
                            const event = new ShowToastEvent({
                                title: 'Error',
                                message: ClaimsAttachEntityMsg,
                                variant: 'error',
                                mode: 'dismissable'
                            });
                            this.dispatchEvent(event);
                        }
                    }).catch(error => {
                        console.log('Error----' + error);
                    });
                    //console.log(`caseid ${caseId} and data = ${JSON.stringify(data)}`);
                } else {
                    this.showTable = true;
                    this.initilizeRecordData();
                }
            }
            else {
                this.showTable = true;
                this.initilizeRecordData();
            }
        }
    }

    @api
    setRecordsOnPage() {
        this.recordsToDisplay = [];
        if (!this.pageSize)
            this.pageSize = 10;

        this.totalPages = Math.ceil(this.filtredNum / this.pageSize);
        this.setPaginationControls();
        for (let i = (this.pageNumber - 1) * this.pageSize; i < this.pageNumber * this.pageSize; i++) {
            if (i === this.filtredNum) break;
            this.recordsToDisplay.push(this.filteredRecords[i]);
        }

        if (this.maxRowSelection === '1') {
            this.totalSelected = 0;
        }

        this.preSelected = [];
        if (this.selectedRecords && this.selectedRecords.length > 0) {
            this.selectedRecords.forEach((item) => {
                if (item.selected)
                    this.preSelected.push(item.Id);
            })
            this.refreshCurrentData = true;
            this.hasPreselectedData = false;
        }
    }

    setPaginationControls() {
        // Previous/Next buttons visibility by Total pages
        if (this.totalPages === 1) {
            this.showPrevious = HIDEDIV;
            this.showNext = HIDEDIV;
        } else if (this.totalPages > 1) {
            this.showPrevious = SHOWDIV;
            this.showNext = SHOWDIV;
        }
        // Previous/Next buttons visibility by Page number
        if (this.pageNumber <= 1) {
            this.pageNumber = 1;
            this.showPrevious = HIDEDIV;
        } else if (this.pageNumber >= this.totalPages) {
            this.pageNumber = this.totalPages;
            this.showNext = HIDEDIV;
        }
        // Previous/Next buttons visibility by Pagination visibility
        if (this.paginationVisibility === false) {
            this.showPrevious = HIDEDIV;
            this.showNext = HIDEDIV;
        }
    }

    handelRowsSelected(selectedRows) {
        this.totalSelected = 0;
        this.pageSelectedRecords = [];
        if (this.maxRowSelection != 1 && this.recordsToDisplay &&
            this.recordsToDisplay.length > 0 &&
            ((selectedRows.length === 0 && !this.refreshCurrentData) || selectedRows.length > 0)) {
            this.recordsToDisplay.forEach((item) => {
                var row = new Object();
                row.Id = item.Id;
                if (selectedRows.includes(item.Id)) {
                    row.selected = true;
                } else {
                    row.selected = false;
                }
                this.pageSelectedRecords.push(row);
            });
        } else if (this.maxRowSelection === 1 && this.recordsToDisplay &&
            this.recordsToDisplay.length > 0 &&
            ((selectedRows.length === 0 && !this.refreshCurrentData) || selectedRows.length > 0)) {
            this.selectedRecords = [];
            this.recordsToDisplay.forEach((item) => {
                var row = new Object();
                row.Id = item.Id;
                if (selectedRows.includes(item.Id)) {
                    row.selected = true;
                } else {
                    row.selected = false;
                }
                this.pageSelectedRecords.push(row);
            });
        }

        // To store previous row Selection
        if (this.selectedRecords.length == 0) {
            this.selectedRecords = this.pageSelectedRecords;
        }
        this.selectedRecords = this.mergeObjectArray(this.selectedRecords, this.pageSelectedRecords, "Id");

        const filterSelected = this.selectedRecords.filter(({
            selected
        }) => selected === true);

        if (this.maxRowSelection === 1 && selectedRows && selectedRows.length > 0) {
            this.totalSelected = 1;
        } else {
            let i = 0;
            this.selectedRecords.forEach(item => {
                if (item.selected) {
                    i++;
                    this.totalSelected = i;
                }
            })
        }
        this.refreshCurrentData = false;
    }

    mergeObjectArray(firstArray, secondArray, prop) {
        var reduced = firstArray.filter(aitem => !secondArray.find(bitem => aitem[prop] === bitem[prop]))
        return reduced.concat(secondArray);
    }

        if (this.sectionName == "Claims" && sortedBy == "totalPaidAmount") {
            let claimsWhichAreNotAccepted = [];
            let claimsWhichAreAccepted = [];
            let claimsAfterSorting = [];
            cloneData.forEach(item => {
                item.totalPaidAmount || item.totalPaidAmount == 0 ? claimsWhichAreAccepted.push(item) : claimsWhichAreNotAccepted.push(item);
            })
            claimsWhichAreAccepted.sort(this.sortBy(sortedBy, sortDirection === 'asc' ? 1 : -1));
            sortDirection === 'asc' ? claimsAfterSorting = [...claimsWhichAreNotAccepted, ...claimsWhichAreAccepted] : claimsAfterSorting = [...claimsWhichAreAccepted, ...claimsWhichAreNotAccepted];
            this.filteredRecords = claimsAfterSorting;
        } else {
            cloneData.sort(this.sortBy(sortedBy, sortDirection === 'asc' ? 1 : -1));
            this.filteredRecords = cloneData;
        }
        this.sortDirection = sortDirection;
        this.sortedBy = sortedBy;
        this.setRecordsOnPage();

    }

    sortBy(field, reverse, primer) {
        const key = primer ?
            function (x) {
                return primer(x[field]);
            } :
            function (x) {
                return x[field];
            };
        return function (a, b) {
            a = key(a) ? key(a) : '';
            b = key(b) ? key(b) : '';
            return reverse * ((a > b) - (b > a));
        };
    }

    invokeWorkspaceAPI(methodName, methodArgs) {
        return new Promise((resolve, reject) => {
            const apiEvent = new CustomEvent("internalapievent", {
                bubbles: true,
                composed: true,
                cancelable: false,
                detail: {
                    category: "workspaceAPI",
                    methodName: methodName,
                    methodArgs: methodArgs,
                    callback: (err, response) => {
                        if (err) {
                            return reject(err);
                        } else {
                            return resolve(response);
                        }
                    }
                }
            });

            window.dispatchEvent(apiEvent);
        });
    }
